```kotlin
private fun fetchSingleRss(urlString: String, category: String): List<NewsArticle> {
    // ... existing code ...

    val impactScore = calculateBasicImpactScore(articleTitle, articleSummary, category)

            impact_score = impactScore  // Calculated, not hardcoded
        )
    ))
}

private fun calculateBasicImpactScore(title: String, summary: String, category: String): Double {
    val text = "$title $summary".lowercase()
    var score = 0.0
    
    // High-impact keywords
    if (text.contains("rate hike") || text.contains("rate cut")) score += 30.0
    if (text.contains("inflation")) score += 20.0
    if (text.contains("gdp")) score += 20.0
    if (text.contains("employment") || text.contains("nfp")) score += 25.0
    if (text.contains("crisis") || text.contains("emergency")) score += 40.0
    
    // Category boost
    when (category) {
        "central-banks" -> score += 20.0
        "forex" -> score += 15.0
        "commodities" -> score += 10.0
    }
    
    return score.coerceIn(0.0, 100.0)
}

private fun extractAssetTags(title: String, summary: String): List<String> {
    val text = "$title $summary".uppercase()
    val tags = mutableListOf<String>()
    
    val currencies = listOf("USD", "EUR", "GBP", "JPY", "CHF", "AUD", "NZD", "CAD")
    val commodities = listOf("GOLD", "SILVER", "OIL", "CRUDE")
    
    currencies.forEach { if (text.contains(it)) tags.add(it) }
    commodities.forEach { if (text.contains(it)) tags.add(it) }
    
    return tags
}
```
